Attack Overview

What Happened?

A China-linked threat actor (Lotus Blossom, aka Billbug/Spring Dragon) compromised the hosting infrastructure of Notepad++ to hijack update traffic. Starting June 2025, selective traffic redirection served tampered updates containing the Chrysalis backdoor to targeted users via the WinGUp (GUP.exe) update mechanism.

Vulnerability: Insufficient update verification in Notepad++ versions prior to v8.8.9

KQL Detection Queries

3.1 GUP.exe Traffic Hijack Detection

Query #1: Detect GUP.exe connecting to malicious infrastructure KQL 
// Detect GUP.exe traffic to known malicious or unknown infrastructure
let MaliciousInfra = dynamic([
    "95.179.213.0",           // Payload delivery
    "61.4.102.97",            // Chrysalis C2
    "59.110.7.32",            // Cobalt Strike
    "124.222.137.114",        // Cobalt Strike
    "api.skycloudcenter.com", // Chrysalis C2
    "api.wiresguard.com"      // Cobalt Strike C2
]);
let LegitimateDomains = dynamic([
    "notepad-plus-plus.org",
    "github.com",
    "githubusercontent.com"
]);
DeviceNetworkEvents
| where Timestamp > ago(30d)
| where InitiatingProcessFileName =~ "gup.exe"
| extend Domain = extract(@"https?://([^/:]+)", 1, RemoteUrl)
| extend IsMalicious = RemoteIP in~ (MaliciousInfra) or Domain in~ (MaliciousInfra)
| extend IsLegitimate = Domain has_any (LegitimateDomains)
| where IsMalicious or not(IsLegitimate)
| project Timestamp, DeviceName, RemoteIP, RemoteUrl, IsMalicious

Query #2: Detect known Chrysalis file hashes KQL

let ChrysalisHashes = dynamic([
    "a511be5164dc1122fb5a7daa3eef9467e43d8458425b15a640235796006590c9",
    "2da00de67720f5f13b17e9d985fe70f10f153da60c9ab1086fe58f069a156924",
    "3bdc4c0637591533f1d4198a72a33426c01f69bd2e15ceee547866f65e26b7ad",
    "f4d829739f2d6ba7e3ede83dad428a0ced1a703ec582fc73a4eee3df3704629a",
    "b4169a831292e245ebdffedd5820584d73b129411546e7d3eccf4663d5fc5be3"

union DeviceFileEvents, DeviceProcessEvents, DeviceImageLoadEvents
| where Timestamp > ago(30d)

3.4 DLL Sideloading Detection

Query #4: Detect BluetoothService.exe sideloading KQL 
// Detect BluetoothService.exe loading DLLs outside Bitdefender paths
let MaliciousHash = "3bdc4c0637591533f1d4198a72a33426c01f69bd2e15ceee547866f65e26b7ad";
DeviceImageLoadEvents
| where Timestamp > ago(30d)
| where InitiatingProcessFileName in~ ("BluetoothService.exe", "BDSubWiz.exe")
    or SHA256 =~ MaliciousHash
| where not(FolderPath has "Program Files\\Bitdefender")
| where not(FolderPath has "Windows\\System32")
| project Timestamp, DeviceName, FileName, FolderPath, SHA256, InitiatingProcessFileName

3.5 TCC Shellcode Execution Detection

Query #5: Detect Tiny-C-Compiler shellcode execution KQL 
// Detect TCC shellcode execution pattern used by Chrysalis
DeviceProcessEvents
| where Timestamp > ago(30d)
| where not(FolderPath startswith "/")  // Windows only
| where
    (ProcessCommandLine has "-nostdlib" and ProcessCommandLine has "-run")
    or (ProcessCommandLine has "conf.c" and ProcessCommandLine has "C:\\")
    or (FileName =~ "svchost.exe" and not(FolderPath has "Windows\\System32"))
    or FolderPath has "ProgramData\\USOShared"
| project Timestamp, DeviceName, FileName, FolderPath, ProcessCommandLine

3.6 Persistence Detection

Query #6: Detect Chrysalis persistence mechanisms KQL 
// Detect Chrysalis-specific persistence indicators
let ChrysalisIndicators = dynamic([
    "BluetoothService.exe",
    "USOShared",
    "ConsoleApplication2.exe",
    "skycloudcenter",
    "wiresguard"

DeviceRegistryEvents
| where Timestamp > ago(30d)
| where RegistryKey has_any (
    "SOFTWARE\\Microsoft\\Windows\\CurrentVersion\\Run",
    "SYSTEM\\CurrentControlSet\\Services"
)
| where RegistryValueData has_any (ChrysalisIndicators)
    or (RegistryValueData has "svchost.exe" and not(RegistryValueData has "System32"))
| project Timestamp, DeviceName, RegistryKey, RegistryValueData
