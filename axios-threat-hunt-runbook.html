Axios NPM Supply Chain RAT

Maintainer Account Hijack & Cross-Platform RAT — Threat Hunt Runbook

Critical Supply Chain Compromise

Axios npm package (~100M+ weekly downloads) compromised via maintainer account hijack. Cross-platform RAT deployed via phantom dependency plain-crypto-js@4.2.1. C2: sfrclak[.]com:8000 (142.11.206.73). Advisories: GHSA-fw8c-xr5c-95f9, MAL-2026-2307.

Attack Overview

What is the Axios Supply Chain Attack?

On March 31, 2026, two malicious versions of the axios npm package (v1.14.1 and v0.30.4) were published after the lead maintainer's npm account (jasonsaayman) was hijacked. The compromised versions inject a phantom dependency (plain-crypto-js@4.2.1) whose sole purpose is to deploy a cross-platform Remote Access Trojan via a postinstall hook. The RAT targets Windows, macOS, and Linux with platform-specific payloads, communicating with C2 infrastructure at sfrclak[.]com:8000 (142.11.206.73, AS54290 Hostwinds LLC). The malicious versions were live for approximately 3 hours before npm removed them.

C2 Infrastructure

C2 Server Summary

The C2 domain sfrclak.com was registered approximately 8 hours before the first malicious package was published. It resolves to 142.11.206.73 hosted on a Hostwinds LLC VPS (AS54290) in Dallas, Texas. The C2 operated over plaintext HTTP on port 8000 with no SSL/TLS certificate. The domain has zero web presence — never indexed by search engines, never archived by the Wayback Machine. The string "sfrclak" has no meaning in any language and appears nowhere as a username or identifier on any platform.

KQL Detection Queries

Deploy these queries in Microsoft Sentinel or Defender Advanced Hunting to detect indicators associated with this threat. 11 queries organized by detection category.

4.1 Network Detection

Query #1 [P1-CRIT]: C2 Network — IP, Domain, Port 8000 KQL 
// Query #1 [P1-CRIT]: C2 Network — IP, Domain, Port 8000
// Table: DeviceNetworkEvents | MITRE: T1071.001, T1571
let C2_IPs = dynamic(["142.11.206.73"]);
let C2_Domains = dynamic(["sfrclak.com", "sfrclak"]);
DeviceNetworkEvents
| where TimeGenerated > ago(30d)
| where RemoteIP in (C2_IPs)
    or RemoteUrl has_any (C2_Domains)
| project TimeGenerated, DeviceName, RemoteIP, RemoteUrl, RemotePort,
    InitiatingProcessFileName, InitiatingProcessCommandLine,
    InitiatingProcessAccountName, InitiatingProcessParentFileName
| sort by TimeGenerated desc
Query #2 [P1-CRIT]: C2 in Azure Firewall / NSG Flow Logs KQL 
// Query #2 [P1-CRIT]: C2 in Azure Firewall / NSG Flow Logs
// Table: AzureDiagnostics | MITRE: T1071.001
AzureDiagnostics
| where TimeGenerated > ago(30d)
| extend DestIP = column_ifexists("destinationAddress_s", ""),
    DestPort = column_ifexists("destinationPort_s", ""),
    SrcIP = column_ifexists("sourceAddress_s", ""),
    Fqdn = column_ifexists("FQDN_s", ""),
    Msg = column_ifexists("msg_s", ""),
    RuleAction = column_ifexists("action_s", ""),
    RuleName = column_ifexists("ruleName_s", "")
| where DestIP == "142.11.206.73"
    or Fqdn has "sfrclak"
    or Msg has "142.11.206.73"
    or Msg has "sfrclak"
| project TimeGenerated, Resource, ResourceGroup, Category,
    Msg, SrcIP, DestIP, DestPort, Fqdn, RuleAction, RuleName
| sort by TimeGenerated desc
Query #3 [P1-CRIT]: DNS Resolution of C2 Domain KQL 
// Query #3 [P1-CRIT]: DNS Resolution of C2 Domain
// Table: DNS_Summarized_Logs_ip_CL | MITRE: T1071.001
DNS_Summarized_Logs_ip_CL
| where TimeGenerated > ago(30d)
| extend Query = column_ifexists("Query_s", column_ifexists("query_s", "")),
    Answer = column_ifexists("Answer_s", column_ifexists("answer_s", "")),
    ClientIP = column_ifexists("ClientIP_s", column_ifexists("clientIP_s", "")),
    QueryType = column_ifexists("QueryType_s", column_ifexists("queryType_s", "")),
    ResponseCode = column_ifexists("ResponseCode_s", column_ifexists("responseCode_s", ""))
| where Query has_any ("sfrclak.com", "sfrclak")
    or Answer has "142.11.206.73"
| project TimeGenerated, ClientIP, Query, QueryType, Answer, ResponseCode
| sort by TimeGenerated desc

4.2 File-Based Detection

Query #4 [P1-CRIT]: RAT File Artifacts — All Platforms KQL 
// Query #4 [P1-CRIT]: RAT File Artifacts — All Platforms
// Table: DeviceFileEvents | MITRE: T1036.005, T1105, T1070.004
DeviceFileEvents
| where TimeGenerated > ago(30d)
| where FileName in~ ("6202033.vbs", "6202033.ps1", "setup.js", "ld.py")
    or FolderPath has_any ("ProgramData\wt.exe", "\6202033.", "plain-crypto-js")
    or (FolderPath has "ProgramData" and FileName endswith ".exe"
        and InitiatingProcessFileName in~ ("node.exe", "npm.cmd"))
| project TimeGenerated, DeviceName, ActionType, FileName, FolderPath,
    SHA256, SHA1, FileSize, InitiatingProcessFileName,
    InitiatingProcessCommandLine, InitiatingProcessAccountName
| sort by TimeGenerated desc

4.3 Process-Based Detection

Query #5 [P2-HIGH]: Node.js / npm Spawning Suspicious Processes KQL 
// Query #5 [P2-HIGH]: Node.js / npm Spawning Suspicious Processes
// Table: DeviceProcessEvents | MITRE: T1059.001, T1059.005, T1204.002
DeviceProcessEvents
| where TimeGenerated > ago(30d)
| where (
    // Attack marker in any command line
    ProcessCommandLine has "6202033"
    // wt.exe (renamed PowerShell) with hidden/bypass flags from ProgramData
    or (FileName =~ "wt.exe" and FolderPath has "ProgramData"
        and ProcessCommandLine has_any ("-w hidden", "-ep bypass"))
    // wscript launching .vbs from Temp (VBScript dropper chain)
    or (FileName in~ ("wscript.exe", "cscript.exe")
        and ProcessCommandLine has "Temp" and ProcessCommandLine has ".vbs")
    // node/npm directly spawning wscript (postinstall -> VBS is abnormal)
    or (InitiatingProcessFileName in~ ("node.exe", "npm.cmd")
        and FileName in~ ("wscript.exe", "cscript.exe"))
    // plain-crypto-js reference in process tree
    or ProcessCommandLine has "plain-crypto-js"
    or InitiatingProcessCommandLine has "plain-crypto-js"
    // T1059.006 — Python RAT on Linux (nohup python3 /tmp/ld.py)
    or (FileName in~ ("python.exe", "python3.exe", "python3")
        and ProcessCommandLine has "/tmp/ld.py")
    // T1059.002 — macOS AppleScript payload (osascript from node)
    or (FileName =~ "osascript" and InitiatingProcessFileName in~ ("node", "npm", "zsh", "bash"))
)
| project TimeGenerated, DeviceName, AccountName, FileName, FolderPath,
    ProcessCommandLine, InitiatingProcessFileName,
    InitiatingProcessCommandLine, InitiatingProcessParentFileName
| sort by TimeGenerated desc
Query #6 [P2-HIGH]: Cloud-Managed Endpoints KQL 
// Query #6 [P2-HIGH]: Same Detection for Cloud-Managed Endpoints
// Table: CloudProcessEvents | MITRE: T1059
CloudProcessEvents
| where TimeGenerated > ago(7d)
| where ProcessCommandLine has_any ("6202033", "plain-crypto-js")
    or (ProcessName =~ "wt.exe" and FolderPath has "ProgramData"
        and ProcessCommandLine has_any ("-w hidden", "-ep bypass"))
    or (ProcessName in~ ("wscript.exe", "cscript.exe")
        and ProcessCommandLine has "Temp" and ProcessCommandLine has ".vbs")
    or (ParentProcessName in~ ("node.exe", "npm.cmd")
        and ProcessName in~ ("wscript.exe", "cscript.exe"))
| project TimeGenerated, AccountName, ProcessName, FolderPath,
    ProcessCommandLine, ParentProcessName
| sort by TimeGenerated desc

        "Get-ChildItem", "dir /s", "tasklist", "systeminfo")
| project TimeGenerated, DeviceName, AccountName, FileName,
    ProcessCommandLine, InitiatingProcessFileName,
    InitiatingProcessFolderPath, InitiatingProcessCommandLine
| sort by TimeGenerated desc
