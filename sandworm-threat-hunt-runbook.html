Attack Overview

What is SANDWORM_MODE?

A self-propagating JavaScript worm discovered in February 2026, distributed through 23 typosquatted npm packages (19 active payload, 4 sleeper). Published by actors using aliases "official334" and "javaorg" with ProtonMail contacts. The worm targets developers' machines and CI/CD environments with a multi-stage attack: Stage 1 loads an obfuscated payload from lib/color-support-engine.min.js (167KB), using XOR + AES-256-GCM encryption. Once active, it harvests SSH keys, cloud credentials (.aws, .azure, Solana wallets), .env files, .npmrc tokens, and API keys for 9 major LLM providers. It spawns password manager CLIs (Bitwarden, 1Password, LastPass) to extract vaults. The worm achieves persistence through git template hooks (pre-commit, post-commit, pre-push) and propagates by injecting malicious GitHub Actions (ci-quality/code-quality-check@v1) into CI workflows. It poisons MCP configurations for Claude Code, Claude Desktop, Cursor, VS Code Continue, and Windsurf. Exfiltration uses dual channels: HTTPS to Cloudflare Workers (pkg-metrics.official334.workers.dev) and DNS tunneling through freefan.net/fanfree.net with a DGA seeded by "sw2025" rotating every 4 hours. A dormant disk wipe capability and polymorphic engine were discovered but are currently disabled.

Attack Chain

1
Initial Compromise
Developer installs typosquatted npm package (e.g., claud-code instead of claude-code).
npm install claud-code
2
Stage 1 Loader
postinstall script executes obfuscated lib/color-support-engine.min.js (167KB). XOR key decrypts Stage 2 payload.
3
Credential Harvesting
Reads SSH keys, .aws/credentials, .npmrc, .env, .config/solana/id.json, .azure/. Spawns bw/op/lpass for password manager vaults. Targets API keys for Anthropic, OpenAI, Mistral, Cohere, Google, and 4 more LLM providers.
4
Git Hook Persistence
Writes malicious hooks to ~/.git-templates/hooks/ (pre-commit, post-commit, pre-push). Sets git config --global init.templateDir to ensure hooks infect every new/cloned repo.
5
CI/CD Injection
Scans for .github/workflows/*.yml and injects ci-quality/code-quality-check@v1 action with pull_request_target trigger. Creates ephemeral workflow formatter_123456789.yml.
6
MCP Poisoning
Drops malicious MCP server configs into ~/.dev-utils/. Modifies config files for Claude Code, Claude Desktop, Cursor, Continue, and Windsurf to load the rogue server.
7
Exfiltration
Dual-channel: HTTPS POST to pkg-metrics.official334.workers.dev/exfil and /drain. DNS tunneling via freefan.net/fanfree.net with DGA rotation.
8
Self-Propagation
Uses stolen npm tokens to publish infected versions of the victim's own packages, spreading the worm to downstream consumers.

KQL Detection Queries

Deploy these queries in Microsoft Sentinel or Defender Advanced Hunting to detect indicators associated with this threat. All 7 queries are self-contained and can be run independently.

3.1 IOC Sweep

Query #1: Known IOC Sweep (packages, domains, alerts, email delivery) KQL 
let MaliciousPackages = dynamic([
    "claud-code","cloude-code","cloude","suport-color","rimarf","naniod",
    "yarsg","opencraw","secp256","veim","hardhta","crypto-locale",
    "detect-cache","parse-compat","crypto-reader-info","format-defaults",
    "locale-loader-pro","node-native-bridge","scan-store",
    "ethres","iru-caches","iruchache","uudi"
]);
let MaliciousDomains = dynamic([
    "pkg-metrics.official334.workers.dev","official334.workers.dev",
    "freefan.net","fanfree.net"
]);
// --- Device process hits (npm install, node exec, package refs) ---
let P = DeviceProcessEvents
| where Timestamp > ago(90d)
| where ActionType == "ProcessCreated"
| where ProcessCommandLine has_any (MaliciousPackages)
    or ProcessCommandLine has_any ("ci-quality/code-quality-check","color-support-engine")
    or ProcessCommandLine has_any ("SANDWORM_MODE","SANDWORM_REGISTRY_URL",
        "SANDWORM_GITHUB_API","SANDWORM_DRAIN_ENDPOINT","SANDWORM_DGA_SEED")
| project Timestamp, AccountSid, AccountName,
          DeviceName,
          Signal="Process_IOC",
          Detail=ProcessCommandLine,
          Source=FileName;
// --- URL click hits (user clicked C2 domain in Teams/email) ---
let U = UrlClickEvents
| where Timestamp > ago(90d)
| where Url has_any (MaliciousDomains)
| project Timestamp, AccountSid="", AccountName=AccountUpn,
          DeviceName="",
          Signal="UrlClick_IOC",
          Detail=Url,
          Source="UrlClickEvents";
// --- Email delivery containing C2 URLs ---
let E = EmailUrlInfo
| where Timestamp > ago(90d)
| where Url has_any (MaliciousDomains)
| join kind=inner (
    EmailEvents
    | where Timestamp > ago(90d)
    | project NetworkMessageId, RecipientEmailAddress, SenderFromAddress, Subject
) on NetworkMessageId
| project Timestamp, AccountSid="", AccountName=RecipientEmailAddress,
          DeviceName="",
          Signal="Email_IOC",
          Detail=strcat(Subject, " | ", Url),
          Source="EmailUrlInfo";
// --- Existing alerts referencing campaign indicators ---
let A = AlertInfo
| where Timestamp > ago(90d)
| where Title has_any (MaliciousPackages)
    or Title has_any (MaliciousDomains)
    or Title has_any ("supply chain","npm","malicious package")
| join kind=inner (
    AlertEvidence
    | project AlertId, EntityType, RemoteUrl, FileName, AccountObjectId
) on AlertId
| project Timestamp, AccountSid=AccountObjectId, AccountName="",
          DeviceName="",
          Signal="Alert_IOC",
          Detail=strcat(Title, " | ", coalesce(RemoteUrl, FileName, "")),
          Source=strcat("Alert:", Severity);
P | union U, E, A
| sort by Timestamp desc

3.2 Credential Harvesting

Query #2: Credential Harvesting via Device Processes KQL 
let NodeProcs = dynamic(["node","node.exe","bun","bun.exe"]);
// --- Process accessing credential files ---
let CredAccess = DeviceProcessEvents
| where Timestamp > ago(30d)
| where ActionType == "ProcessCreated"
| where FileName in~ (NodeProcs) or InitiatingProcessFileName in~ (NodeProcs)
// Exclude find/fd â€” they scan directories and match on path fragments, not cred access
| where FileName !in~ ("find","find.exe","fd","fd.exe","locate","mdfind",
                        "rg","rg.exe","grep","grep.exe","ag","ag.exe")
| where ProcessCommandLine has_any (
    ".ssh/id_rsa",".ssh/id_ed25519",".ssh/id_ecdsa",
    ".aws/credentials","accessTokens.json",
    "application_default_credentials.json",
    ".config/solana/id.json",
    ".npmrc",".azure/",".config/gcloud")
    // .env matched tightly: require path separator before it to avoid false hits
    or ProcessCommandLine matches regex @"[/\\]\.env\b"
| project Timestamp, AccountSid, AccountName,
          DeviceName,
          Signal="CredFile_Access",
          Detail=ProcessCommandLine,
          Source=FileName;
// --- Password manager CLI spawned by node ---
let PwMgr = DeviceProcessEvents
| where Timestamp > ago(30d)
| where ActionType == "ProcessCreated"
| where InitiatingProcessFileName in~ (NodeProcs)
| where FileName in~ ("bw","bw.exe","op","op.exe","lpass","lpass.exe")
| project Timestamp, AccountSid, AccountName,
          DeviceName,
          Signal="PasswordMgr_Spawn",
          Detail=ProcessCommandLine,
          Source=InitiatingProcessFileName;
// --- GitHub/LLM tokens exposed in command lines ---
let TokenLeak = DeviceProcessEvents
| where Timestamp > ago(30d)
| where ActionType == "ProcessCreated"
| where ProcessCommandLine matches regex @"gh[po]_[A-Za-z0-9_]{36,}"
    or ProcessCommandLine matches regex @"github_pat_[A-Za-z0-9_]{22,}"
    or ProcessCommandLine has_any (
        "ANTHROPIC_API_KEY","OPENAI_API_KEY","MISTRAL_API_KEY",
        "COHERE_API_KEY","TOGETHER_API_KEY","FIREWORKS_API_KEY",
        "REPLICATE_API_TOKEN","GROK_API_KEY","GOOGLE_API_KEY",
        "sk-ant-","sk-proj-")
| where FileName !in~ ("git","git.exe","gh","gh.exe")
| project Timestamp, AccountSid, AccountName,
          DeviceName,
          Signal="Token_Leak",
          Detail=replace_regex(ProcessCommandLine,
              @"(gh[po]_|github_pat_|sk-[a-z]+-)[A-Za-z0-9_-]+",
              @"\1[REDACTED]"),
          Source=FileName;
CredAccess | union PwMgr, TokenLeak
| sort by Timestamp desc
