Attack Overview

What is Void Manticore?

Void Manticore (Storm-0842 / Red Sandstorm / Banished Kitten / COBALT MYSTIQUE) is an Iranian state-sponsored destructive threat actor operated by the Ministry of Intelligence and Security (MOIS), Internal Security Deputy, Counter-Terrorism Division. Active since 2008, the group operates under multiple hacktivist personas including Handala Hack (Israel/Global), Karma (Israel), and Homeland Justice (Albania). The group works in a “one-two punch” model with Scarred Manticore (Storm-0861) which provides initial access after 12+ months of espionage, before handing off to Void Manticore for destruction. On March 11, 2026, Handala executed Operation Epic Fury against Stryker Corporation, weaponizing Microsoft Intune’s Remote Wipe feature to destroy approximately 200,000 devices across 79 countries without deploying any malware.

KQL Detection Queries

4.1 Identity & Access Detection

Q1: Admin Sign-ins from Threat IP Ranges
Table: EntraIdSignInEvents  |  MITRE: T1078.004 / T1110
KQL 
let ThreatIPs = "str">dynamic([
    "107.189.19.52", "82.25.35.25", "31.57.35.223",
    "146.185.219.235", "31.192.237.207",
    "64.176.169.22", "64.176.172.235", "64.176.172.165",
    "64.176.173.77", "64.176.172.101"
]);
EntraIdSignInEvents
| where TimeGenerated > ago(30d)
| where IPAddress in (ThreatIPs)
    or IPAddress startswith "188.92.255."
    or IPAddress startswith "209.198.131."
    or IPAddress startswith "169.150.227."
    or IPAddress startswith "149.88.26."
| summarize
    SuccessCount = countif(ErrorCode == 0),
    FailCount = countif(ErrorCode != 0),
    Accounts = make_set(AccountUpn, 20),
    Apps = make_set(Application, 10)
    by IPAddress, bin(TimeGenerated, 1h)
| project TimeGenerated, IPAddress, SuccessCount, FailCount, Accounts, Apps
| sort by SuccessCount desc, FailCount desc
Q2: NON-PIM Privileged Role Assignment + Intune OAuth Grants
Table: AuditLogs  |  MITRE: T1098 / T1078.004
KQL 
AuditLogs
| where TimeGenerated > ago(30d)
| where
    (OperationName in (
        "Add member to role",
        "Add eligible member to role")
    and not(OperationName has "PIM")
    and not(InitiatedBy has "MS-PIM")
    and TargetResources has_any (
        "Global Administrator",
        "Intune Administrator",
        "Intune Service Administrator",
        "Privileged Role Administrator",
        "Cloud Device Administrator"))
    or (OperationName has_any (
        "Consent to application",
        "Add OAuth2PermissionGrant")
    and TargetResources has_any (
        "DeviceManagementManagedDevices.PrivilegedOperations.All",
        "DeviceManagementManagedDevices.ReadWrite.All"))
| project TimeGenerated, OperationName, InitiatedBy, TargetResources,
    Result, CorrelationId
| sort by TimeGenerated desc

4.2 Credential & Evasion Detection

Q3: LSASS Credential Dump via comsvcs.dll
Table: DeviceProcessEvents  |  MITRE: T1003.001
KQL 
DeviceProcessEvents
| where TimeGenerated > ago(30d)
| where FileName =~ "rundll32.exe"

| project TimeGenerated, DeviceName, ProcessCommandLine, AccountName,
    AccountDomain, InitiatingProcessFileName,
    InitiatingProcessCommandLine, ProcessId
| sort by TimeGenerated desc
Q6: Defender Tampering via Registry
Table: DeviceRegistryEvents  |  MITRE: T1562.001
KQL 
DeviceRegistryEvents
| where TimeGenerated > ago(30d)
| where RegistryKey has "Windows Defender"
| where RegistryValueName in~ (
    "TamperProtection",
    "DisableRealtimeMonitoring",
    "DisableAntiSpyware",
    "DisableBehaviorMonitoring"
)
| where RegistryValueData in ("0", "1")
| where not(InitiatingProcessFileName in~ (
    "MsMpEng.exe", "svchost.exe", "TiWorker.exe"
))
| project TimeGenerated, DeviceName, RegistryKey, RegistryValueName,
    RegistryValueData, InitiatingProcessFileName,
    InitiatingProcessCommandLine, InitiatingProcessAccountName
| sort by TimeGenerated desc

4.3 Lateral Movement & C2

Q4: AD Enumeration - ADRecon + Bulk LDAP
Table: DeviceProcessEvents + IdentityQueryEvents  |  MITRE: T1087.002 / T1018
KQL 
let ADReconHits =
    DeviceProcessEvents
    | where TimeGenerated > ago(30d)
    | where FileName in~ ("powershell.exe", "pwsh.exe")
    | where ProcessCommandLine has_any (
        "dra.ps1", "ADRecon", "Invoke-ADRecon",
        "Get-ADUser", "Get-ADComputer", "Get-ADGroup",
        "Get-DomainUser", "Get-DomainComputer")
    | project TimeGenerated, DeviceName, Source = "ADRecon_Process",
        Detail = ProcessCommandLine, AccountName;
let BulkLDAP =
    IdentityQueryEvents
    | where TimeGenerated > ago(30d)
    | where QueryType has_any ("LDAP", "Samr")
    | summarize QueryCount = count()
        by AccountName, DeviceName, QueryType, bin(TimeGenerated, 1h)
    | where QueryCount >= 100
    | project TimeGenerated, DeviceName, Source = "Bulk_LDAP",
        Detail = strcat(QueryType, " x", QueryCount), AccountName;
union ADReconHits, BulkLDAP
| sort by TimeGenerated desc

DeviceLogonEvents

| where LogonType == "RemoteInteractive"
| where RemoteDeviceName in~ (AttackerHostnames)

    RemoteDeviceName, RemoteIP, LogonType
| sort by TimeGenerated desc
Q7: NetBird + VeraCrypt + AutoIt Tool Stack
Table: DeviceProcessEvents  |  MITRE: T1572 / T1486 / T1055
KQL 
DeviceProcessEvents
| where TimeGenerated > ago(30d)
| where
    FileName has_any ("netbird", "netbird.exe")
    or FileName has_any ("veracrypt", "VeraCrypt.exe", "VeraCrypt Setup")
    or FileName =~ "autoit3.exe"
    or ProcessCommandLine has ".a3x"
    or (FileName =~ "regasm.exe"
        and not(ProcessCommandLine has ".dll")
        and not(ProcessCommandLine has ".exe"))
| project TimeGenerated, DeviceName, FileName, ProcessCommandLine,
    FolderPath, AccountName, InitiatingProcessFileName,
    InitiatingProcessCommandLine
| sort by TimeGenerated desc

4.5 Destruction Detection

Q9: Intune Mass Wipe via Graph API
Table: MicrosoftGraphActivityLogs  |  MITRE: T1531 / T1072
KQL 
MicrosoftGraphActivityLogs
| where TimeGenerated > ago(30d)
| where RequestUri has "/deviceManagement/managedDevices"
| where RequestUri has_any (
    "/wipe", "/retire", "/factoryReset",
    "/resetPasscode", "/delete"
)
| where RequestMethod in ("POST", "DELETE")
| summarize
    ActionCount = count(),
    Endpoints = make_set(RequestUri, 20)
    by UserId, IPAddress, UserAgent, bin(TimeGenerated, 10m)
| project TimeGenerated, UserId, IPAddress, UserAgent,
    ActionCount, Endpoints
| sort by ActionCount desc
Q10: Wiper Preparation - Shadow Copy + Recovery Disable
Table: DeviceProcessEvents  |  MITRE: T1490 / T1561.002
KQL 
DeviceProcessEvents
| where TimeGenerated > ago(30d)
| where
    (ProcessCommandLine has "vssadmin" and ProcessCommandLine has "delete"
        and ProcessCommandLine has "shadows")
    or (ProcessCommandLine has "wmic" and ProcessCommandLine has "shadowcopy"
        and ProcessCommandLine has "delete")
    or (FileName =~ "bcdedit.exe"
        and ProcessCommandLine has_any ("recoveryenabled", "bootstatuspolicy"))
    or (FileName =~ "wbadmin.exe" and ProcessCommandLine has "delete"
        and ProcessCommandLine has "catalog")
| project TimeGenerated, DeviceName, FileName, ProcessCommandLine,
    AccountName, InitiatingProcessFileName,
    InitiatingProcessCommandLine
| sort by TimeGenerated desc
