Attack Overview

What is the Rekoobe Backdoor Campaign?

A malicious Go module (github.com/xinfeisoft/crypto) impersonating the legitimate golang.org/x/crypto library has been discovered deploying the Rekoobe Linux backdoor through a supply chain attack. The module hooks the ReadPassword() function in ssh/terminal/terminal.go to silently capture terminal passwords, exfiltrates them to attacker infrastructure, establishes SSH persistence, disables host firewalls, and deploys the Rekoobe backdoor — a sophisticated ELF implant attributed to APT31 (Zirconium).

Attack Chain

1
Initial Access
Developer or CI/CD pipeline pulls github.com/xinfeisoft/crypto as a Go dependency
2
Execution
Backdoored ReadPassword() function activates during terminal password prompts
3
Credential Theft
Captured passwords exfiltrated to http://img.spoolsv.cc/seed.php
4
Staging
Stager shell script (snn50.txt) downloaded and executed
5
Persistence
Attacker SSH key appended to /home/ubuntu/.ssh/authorized_keys
6
Defense Evasion
iptables default policies set to ACCEPT, disabling firewall
7
Payload Delivery
sss.mp5 (loader) and 555.mp5 (Rekoobe) downloaded from img.spoolsv.cc
8
C2 Establishment
sss.mp5 establishes encrypted TCP to 154.84.63.184:443
9
Backdoor Active
555.mp5 (Rekoobe) provides full backdoor; masquerades as /bin/bash

Terminal passwords captured via hooked ReadPassword()

Rekoobe ELF with encrypted C2 on TCP 443

Attacker SSH key injected into authorized_keys

KQL Detection Queries

Deploy these queries in Microsoft Sentinel or Defender Advanced Hunting to detect indicators associated with this threat.

3.1 IOC-Based Detection

Query #1: C2 IP Connection Detection (DeviceNetworkEvents) KQL 
// Detect connections to known Rekoobe C2 infrastructure
let ThreatIPs = dynamic(["154.84.63.184"]);
DeviceNetworkEvents
| where TimeGenerated > ago(30d)
| where RemoteIP in (ThreatIPs)
| project TimeGenerated, DeviceName, RemoteIP, RemotePort, Protocol,
    LocalIP, LocalPort, InitiatingProcessFileName, InitiatingProcessCommandLine,
    InitiatingProcessAccountName, InitiatingProcessParentFileName
| sort by TimeGenerated desc
Query #2: Malicious Domain Connections (DeviceNetworkEvents) KQL 
// Detect connections to attacker-controlled domains
let ThreatDomains = dynamic(["spoolsv.cc", "spoolsv.net", "img.spoolsv.cc", "img.spoolsv.net"]);
DeviceNetworkEvents
| where TimeGenerated > ago(30d)
| where RemoteUrl has_any (ThreatDomains)
| project TimeGenerated, DeviceName, RemoteUrl, RemoteIP, RemotePort,
    InitiatingProcessFileName, InitiatingProcessCommandLine,
    InitiatingProcessAccountName
| sort by TimeGenerated desc
Query #3: File Hash IOC Matching (DeviceFileEvents) KQL 
// Match file hashes against known Rekoobe payload hashes
let ThreatSHA256 = dynamic([
    "4afdb3f5914beb0ebe3b086db5a83cef1d3c3c4312d18eff672dd0f6be2146bc",
    "8b0ec8d0318347874e117f1aed1b619892a7547308e437a20e02090e5f3d2da6"
]);
let ThreatMD5 = dynamic(["4be68d5690e8df506583e86e42fbb934"]);
let ThreatSHA1 = dynamic(["590dc30098a1bedf28e1717ded91cbfed72ae06d"]);
DeviceFileEvents
| where TimeGenerated > ago(30d)
| where SHA256 in (ThreatSHA256) or MD5 in (ThreatMD5) or SHA1 in (ThreatSHA1)
| project TimeGenerated, DeviceName, FileName, FolderPath, SHA256, SHA1, MD5,
    FileSize, InitiatingProcessFileName, InitiatingProcessCommandLine, ActionType
| sort by TimeGenerated desc
Query #4: Suspicious .mp5 Payload Files (DeviceFileEvents) KQL 
// Detect suspicious .mp5 payload files and stager scripts
let SuspiciousFiles = dynamic(["sss.mp5", "555.mp5", "snn50.txt", "seed.php"]);
DeviceFileEvents
| where TimeGenerated > ago(30d)
| where FileName in~ (SuspiciousFiles)
    or FileName endswith ".mp5"
| project TimeGenerated, DeviceName, FileName, FolderPath, SHA256, FileSize,
    ActionType, InitiatingProcessFileName, InitiatingProcessCommandLine
| sort by TimeGenerated desc
Query #5: DNS Queries to Threat Infrastructure (DNS_Summarized_Logs_ip_CL) KQL 
// Detect DNS resolution attempts to attacker domains
let ThreatDomains = dynamic(["spoolsv.cc", "spoolsv.net", "img.spoolsv.cc", "img.spoolsv.net"]);
DNS_Summarized_Logs_ip_CL
| where TimeGenerated > ago(30d)
| where DnsQuery_s has_any (ThreatDomains)
| project TimeGenerated, SrcIpAddr_s, DnsQuery_s, EventResultDetails_s, DnsResponseName_s
| sort by TimeGenerated desc
Query #6: TI Feed Correlation (ThreatIntelIndicators) KQL 
// Correlate with Threat Intelligence indicator feeds
let ThreatIPs = dynamic(["154.84.63.184"]);
let ThreatDomains = dynamic(["spoolsv.cc", "spoolsv.net", "img.spoolsv.cc", "img.spoolsv.net"]);
ThreatIntelIndicators
| where TimeGenerated > ago(30d)
| where IsActive == true and IsDeleted == false
| where (ObservableKey == "ipv4-addr" and ObservableValue in (ThreatIPs))
    or (ObservableKey in ("domain-name", "url") and ObservableValue has_any (ThreatDomains))
| project TimeGenerated, ObservableKey, ObservableValue, Confidence,
    Pattern, Tags, SourceSystem, ValidFrom, ValidUntil
| sort by TimeGenerated desc
Query #7: Azure Firewall/NSG C2 Detection (AzureDiagnostics) KQL 
// Detect C2 traffic in Azure Firewall and NSG logs
let ThreatIPs = dynamic(["154.84.63.184"]);
let ThreatDomains = dynamic(["spoolsv.cc", "spoolsv.net", "img.spoolsv.cc", "img.spoolsv.net"]);
AzureDiagnostics
| where TimeGenerated > ago(30d)
| where ResourceType in ("AZUREFIREWALLS", "NETWORKSECURITYGROUPS")
| where msg_s has_any (ThreatIPs)
    or msg_s has_any (ThreatDomains)
    or DestinationIp_s in (ThreatIPs)
    or Fqdn_s has_any (ThreatDomains)
    or TargetUrl_s has_any (ThreatDomains)
| project TimeGenerated, ResourceType, Resource, Category, msg_s,
    SourceIP, DestinationIp_s, DestinationPort_d, Fqdn_s, TargetUrl_s,
    Action_s, Policy_s, RuleCollection_s, Rule_s, Protocol_s
| sort by TimeGenerated desc

3.2 Behavioral Detection

Query #8: SSH Authorized Keys Modification (DeviceFileEvents) KQL 
// Detect modifications to SSH authorized_keys files
DeviceFileEvents
| where TimeGenerated > ago(30d)
| where FolderPath has ".ssh"
    and FileName == "authorized_keys"
| where ActionType in ("FileCreated", "FileModified")
| project TimeGenerated, DeviceName, FolderPath, FileName, ActionType,
    InitiatingProcessFileName, InitiatingProcessCommandLine,
    InitiatingProcessAccountName, InitiatingProcessParentFileName
| sort by TimeGenerated desc
Query #10: Suspicious Curl/Wget Downloads (DeviceProcessEvents) KQL 
// Detect suspicious download tool usage with threat-related patterns
let SuspiciousPatterns = dynamic([".mp5", "seed.php", "snn50.txt", "spoolsv"]);
DeviceProcessEvents
| where TimeGenerated > ago(30d)
| where FileName in~ ("curl", "wget", "python", "python3")
| where ProcessCommandLine has_any (SuspiciousPatterns)
| project TimeGenerated, DeviceName, FileName, ProcessCommandLine,
    AccountName, InitiatingProcessFileName, InitiatingProcessCommandLine
| sort by TimeGenerated desc
Query #11: Go Module Supply Chain Indicators (DeviceProcessEvents) KQL 
// Detect usage of malicious Go module in build processes
let MaliciousModules = dynamic(["xinfeisoft/crypto", "xinfeisoft"]);
DeviceProcessEvents
| where TimeGenerated > ago(30d)
| where FileName in~ ("go", "go.exe")
    or ProcessCommandLine has "go "
| where ProcessCommandLine has_any (MaliciousModules)
| project TimeGenerated, DeviceName, FileName, ProcessCommandLine,
    AccountName, FolderPath, InitiatingProcessFileName
| sort by TimeGenerated desc

3.3 Cross-Table Correlation

Query #14: File Download Then Execution (DeviceFileEvents + DeviceProcessEvents) KQL 
// Correlate file drops with subsequent process execution
let SuspiciousExtensions = dynamic([".mp5"]);
let SuspiciousNames = dynamic(["sss.mp5", "555.mp5", "snn50.txt"]);
let FileDrops = DeviceFileEvents
| where TimeGenerated > ago(30d)
| where FileName in~ (SuspiciousNames) or FileName endswith ".mp5"
| project FileDropTime = TimeGenerated, DeviceName, FileName, FolderPath, SHA256,
    DroppedBy = InitiatingProcessFileName;
let ProcessExec = DeviceProcessEvents
| where TimeGenerated > ago(30d)
| where ProcessCommandLine has_any (SuspiciousNames)
    or ProcessCommandLine has ".mp5"
    or FileName in~ (SuspiciousNames)
| project ExecTime = TimeGenerated, DeviceName, FileName, ProcessCommandLine,
    AccountName, InitiatingProcessFileName;
FileDrops
| join kind=inner ProcessExec on DeviceName
| where ExecTime >= FileDropTime and ExecTime <= datetime_add('hour', 1, FileDropTime)
| project FileDropTime, ExecTime, DeviceName, DroppedBy, FileName,
    ProcessCommandLine, AccountName
| sort by FileDropTime desc
Query #15: Process Spawning C2 Connection (DeviceProcessEvents + DeviceNetworkEvents) KQL 
// Correlate suspicious processes with outbound C2 connections
let ThreatIPs = dynamic(["154.84.63.184"]);
let ThreatDomains = dynamic(["spoolsv.cc", "spoolsv.net", "img.spoolsv.cc", "img.spoolsv.net"]);
let SuspiciousProcs = DeviceProcessEvents
| where TimeGenerated > ago(30d)
| where ProcessCommandLine has_any ("iptables", ".mp5", "snn50", "spoolsv", "xinfeisoft")
    or FileName endswith ".mp5"
| project ProcTime = TimeGenerated, DeviceName, ProcessId, FileName,
    ProcessCommandLine, AccountName;
let C2Connections = DeviceNetworkEvents
| where TimeGenerated > ago(30d)
| where RemoteIP in (ThreatIPs) or RemoteUrl has_any (ThreatDomains)
| project NetTime = TimeGenerated, DeviceName, InitiatingProcessId = InitiatingProcessId,
    RemoteIP, RemotePort, RemoteUrl, InitiatingProcessFileName;
SuspiciousProcs
| join kind=inner C2Connections on DeviceName
| where NetTime >= ProcTime and NetTime <= datetime_add('hour', 2, ProcTime)
| project ProcTime, NetTime, DeviceName, FileName, ProcessCommandLine,
    RemoteIP, RemotePort, RemoteUrl, AccountName
| sort by ProcTime desc
