Attack Overview

What is Shai-Hulud 2.0?

Shai-Hulud 2.0 ("The Second Coming") is a sophisticated NPM supply chain attack discovered on November 24, 2025. The attacker compromised 796+ packages across major organizations including Zapier, PostHog, Postman, ENS Domains, AsyncAPI, Voiceflow, and others.

Attack Chain

1
Initial Infection Vector
Victim installs compromised package
npm install compromised-package
2
Hook Execution
preinstall/postinstall hook executes setup_bun.js
3
Runtime Download
Downloads Bun runtime from bun.sh/install
4
Payload Execution
Executes bun_environment.js (10MB+ obfuscated payload)
5
Credential Scanning
Downloads & runs TruffleHog — Scans filesystem for secrets
6
Data Exfiltration
Exfiltrates credentials via GitHub API
7
Destructive Payload
On failure: WIPES HOME DIR
rm -rf $HOME / shred

Impact

796 Compromised Packages

132M+ monthly downloads affected

Credential Theft

npm, GitHub, AWS, GCP, Azure, SSH keys

Destructive Payload

Wipes home directory on auth failure

Persistence

Registers malicious GitHub Actions runners

KQL Detection Queries

3.1 File-Based Detection

Query #1: Malicious File Detection KQL 
// Detect Shai-Hulud malware files
DeviceFileEvents
| where Timestamp > ago(7d)
| where FileName in~ (
    "setup_bun.js",
    "bun_environment.js",
    "actionsSecrets.json",
    "truffleSecrets.json",
    "trufflehog_output.json",
    "cloud.json",
    "contents.json",
    "environment.json"
)
| where ActionType in ("FileCreated", "FileModified")
| project
    Timestamp,
    DeviceName,
    FileName,
    FolderPath,
    FileSize,
    SHA256,
    InitiatingProcessFileName,
    InitiatingProcessCommandLine,
    Severity = case(
        FileName in~ ("setup_bun.js", "bun_environment.js", "actionsSecrets.json"), "CRITICAL",
        FileName in~ ("truffleSecrets.json", "trufflehog_output.json"), "CRITICAL",
        "HIGH"
    )
| order by Timestamp desc
Query #2: Suspicious Package Install Behavior KQL 
// Detect suspicious behavior during npm lifecycle scripts (behavioral detection)
DeviceProcessEvents
| where Timestamp > ago(7d)
| where InitiatingProcessFileName in~ ("npm", "npm.cmd", "yarn", "yarn.cmd", "pnpm", "pnpm.cmd")
| where InitiatingProcessCommandLine has "install"
| where FileName in~ ("node", "node.exe", "bun", "bun.exe", "curl", "wget", "powershell.exe", "cmd.exe")
| where ProcessCommandLine has_any (
    "bun.sh",
    "eval",
    "base64",
    "trufflehog",
    "webhook.site",
    "api.github.com"
)
| project
    Timestamp,
    DeviceName,
    AccountName,
    ProcessCommandLine,
    InitiatingProcessCommandLine,
    Severity = "CRITICAL",
    Description = "Suspicious behavior during package installation"
| order by Timestamp desc
Query #3: Large Obfuscated Payload Detection KQL 
// Detect bun_environment.js payload (typically 10MB+)
DeviceFileEvents
| where Timestamp > ago(7d)
| where FileName =~ "bun_environment.js"
| where ActionType in ("FileCreated", "FileModified")
| extend FileSizeMB = FileSize / 1048576.0
| project
    Timestamp,
    DeviceName,
    AccountName,
    FileName,
    FolderPath,
    FileSizeMB = round(FileSizeMB, 2),
    SHA256,
    Severity = case(
        FileSizeMB > 5, "CRITICAL - Obfuscated Payload",
        FileSizeMB > 1, "HIGH - Suspicious Size",
        "MEDIUM - Review Required"
    )
| order by Timestamp desc

3.2 Process-Based Detection

Query #4: Bun Installation from NPM Context KQL 
// Detect Bun runtime installation from npm/node context
DeviceProcessEvents
| where Timestamp > ago(7d)
| where ProcessCommandLine has_any ("bun.sh/install", "bun.sh\\install")
| extend
    IsFromNodeProcess = InitiatingProcessFileName in~ ("node", "node.exe", "npm", "npm.cmd"),
    IsFromPackageInstall = InitiatingProcessCommandLine has_any ("install", "preinstall", "postinstall")
| project
    Timestamp,
    DeviceName,
    AccountName,
    ProcessCommandLine,
    InitiatingProcessFileName,
    InitiatingProcessCommandLine,
    Confidence = case(
        IsFromNodeProcess and IsFromPackageInstall, "HIGH - Confirmed Attack Pattern",
        IsFromNodeProcess, "MEDIUM - Suspicious",
        "LOW - Review Context"
    ),
    Severity = "CRITICAL"
| order by Timestamp desc
Query #5: TruffleHog Execution Detection KQL 
// Detect TruffleHog credential scanning

| where Timestamp > ago(7d)
| where ProcessCommandLine has_any (
    "trufflehog",
    "trufflesecurity",
    "--json --no-update",
    "secret-scanner",
    "credential-scan"
)
or FileName in~ ("trufflehog", "trufflehog.exe")
| project
    Timestamp,
    DeviceName,
    AccountName,
    FileName,
    ProcessCommandLine,
    InitiatingProcessFileName,
    InitiatingProcessCommandLine,
    Severity = "CRITICAL",
    Description = "TruffleHog credential scanning detected"
| order by Timestamp desc
Query #6: Suspicious Package.json Script Execution KQL 
// Detect malicious preinstall/postinstall scripts
DeviceProcessEvents
| where Timestamp > ago(7d)
| where FileName in~ ("node", "node.exe", "npm", "npm.cmd", "sh", "bash", "cmd.exe", "powershell.exe")
| where ProcessCommandLine has_any (
    "setup_bun.js",
    "bun_environment.js",
    "preinstall",
    "postinstall"
)
| where ProcessCommandLine has_any (
    "curl", "wget", "eval", "base64", "bun.sh"
)
| project
    Timestamp,
    DeviceName,
    AccountName,
    FileName,
    ProcessCommandLine,
    InitiatingProcessCommandLine,
    Severity = case(
        ProcessCommandLine has_any ("setup_bun.js", "bun_environment.js"), "CRITICAL",
        ProcessCommandLine has_any ("curl|sh", "wget|sh"), "CRITICAL",
        ProcessCommandLine has "eval", "HIGH",
        "MEDIUM"
    )
| order by Timestamp desc
Query #7: Destructive Payload Detection (Home Directory Wipe) KQL 
// Detect home directory wipe attempts
DeviceProcessEvents
| where Timestamp > ago(7d)
| where ProcessCommandLine has_any (
    "rm -rf $HOME", "rm -rf ~", "rm -rf /home/",
    "rmdir /s /q %USERPROFILE%",
    "Remove-Item -Recurse -Force $env:USERPROFILE",
    "find \"$HOME\" -type f -writable -user",
    "xargs -0 -r shred"
)
| project
    Timestamp,
    DeviceName,
    AccountName,
    DestructiveCommand = ProcessCommandLine,
    InitiatingProcess = InitiatingProcessCommandLine,
    Severity = "CRITICAL",
    Description = "Home directory destruction attempt detected"
| order by Timestamp desc

3.3 Network-Based Detection

Query #8: GitHub API Exfiltration Pattern KQL 
// Detect credential upload to GitHub
DeviceNetworkEvents
| where Timestamp > ago(7d)
| where RemoteUrl has "api.github.com"
| where RemoteUrl has_any ("/repos/", "/contents/", "/user")
| where InitiatingProcessFileName in~ ("node", "node.exe", "bun", "bun.exe")
| summarize
    CallCount = count(),
    UniqueEndpoints = dcount(RemoteUrl),
    Endpoints = make_set(RemoteUrl, 10),
    FirstSeen = min(Timestamp),
    LastSeen = max(Timestamp)
by DeviceName, InitiatingProcessFileName
| where CallCount > 5 or UniqueEndpoints > 3
| extend Severity = case(
    CallCount > 20, "CRITICAL",
    CallCount > 10, "HIGH",
    "MEDIUM"
)
| order by CallCount desc
Query #9: TruffleHog Binary Download Detection KQL 
// Detect TruffleHog binary download
DeviceNetworkEvents
| where Timestamp > ago(7d)
| where RemoteUrl has_any (
    "github.com/trufflesecurity/trufflehog",
    "releases/download"
)
| where RemoteUrl has "trufflehog"
| project
    Timestamp,
    DeviceName,
    AccountName,
    RemoteUrl,
    InitiatingProcessFileName,
    InitiatingProcessCommandLine,
    Severity = "CRITICAL",
    Description = "TruffleHog binary download detected"
| order by Timestamp desc
Query #10: GitHub Repo Creation Detection KQL

3.4 Credential Access Detection

Query #12: Sensitive File Access by Node/Bun KQL 
// Detect node/bun accessing credential files
DeviceFileEvents
| where Timestamp > ago(7d)
| where InitiatingProcessFileName in~ ("node", "node.exe", "bun", "bun.exe")
| where FileName in~ (
    ".npmrc",
    ".env",
    "credentials",
    "config",
    ".netrc",
    "id_rsa",
    "id_ed25519",
    ".aws",
    ".azure"
)
or FolderPath has_any (".aws", ".ssh", ".config/gcloud", ".azure")
| project
    Timestamp,
    DeviceName,
    AccountName,
    FileName,
    FolderPath,
    ActionType,
    InitiatingProcessCommandLine,
    Severity = case(
        FileName has_any ("id_rsa", "id_ed25519", "credentials"), "CRITICAL",
        FolderPath has_any (".aws", ".ssh"), "CRITICAL",
        FileName in~ (".npmrc", ".env"), "HIGH",
        "MEDIUM"
    )
| order by Timestamp desc

3.5 GitHub Actions Detection

Query #13: Malicious GitHub Workflow Detection KQL 
// Detect malicious GitHub Actions workflows
DeviceFileEvents
| where Timestamp > ago(7d)
| where FolderPath has ".github"
| where FileName endswith ".yml" or FileName endswith ".yaml"
| where ActionType in ("FileCreated", "FileModified")
| project
    Timestamp,
    DeviceName,
    AccountName,
    FileName,
    FolderPath,
    ActionType,
    SHA256,
    Severity = case(
        FileName startswith "formatter_", "CRITICAL - Malicious formatter workflow",
        FileName in~ ("discussion.yaml", "discussion.yml"), "CRITICAL - Malicious discussion workflow",
        "Review Required"
    )
| where Severity startswith "CRITICAL"
| order by Timestamp desc

)
| project

    DeviceName,
    AccountName,
    ProcessCommandLine,
    Severity = "MEDIUM",
